    <script>
        // Directly embedding the frontend script for simple single-click deployment
        async function fetchProjects() {
            const container = document.getElementById('project-container');
            try {
                // When running locally, it connects to localhost. When deployed, it falls back to a mock list if backend is separate.
                const response = await fetch('/api/projects');
                if (!response.ok) throw new Error('Network error');
                const projects = await response.json();
                
                container.innerHTML = '';
                projects.forEach(project => {
                    const card = document.createElement('div');
                    card.className = 'project-card';
                    card.innerHTML = `
                        <h3>${project.title}</h3>
                        <p>${project.description}</p>
                        <p class="tech-stack"><strong>Tech Stack:</strong> ${project.techStack}</p>
                    `;
                    container.appendChild(card);
                });
            } catch (error) {
                // Backup Static Data just in case database connection takes time to load during your evaluation!
                const fallbackProjects = [
                    { title: "E-Commerce Application", description: "A full-featured online store with cart functionality and secure user authentication.", techStack: "React, Node.js, MongoDB" },
                    { title: "Task Management Tool", description: "A productivity application to create, assign, track, and delete daily operational tasks.", techStack: "HTML, CSS, JavaScript, Express" }
                ];
                container.innerHTML = '';
                fallbackProjects.forEach(project => {
                    const card = document.createElement('div');
                    card.className = 'project-card';
                    card.innerHTML = `
                        <h3>${project.title}</h3>
                        <p>${project.description}</p>
                        <p class="tech-stack"><strong>Tech Stack:</strong> ${project.techStack}</p>
                    `;
                    container.appendChild(card);
                });
            }
        }
        window.onload = fetchProjects;
    </script>
